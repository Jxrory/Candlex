### Task 2: Create NewsSource

**Files:**
- Create: `app/sources/news.py`

- [ ] **Step 1: Create the source file**

```python
import logging

import akshare as ak
import pandas as pd

from app.ak_trace import trace_akshare

logger = logging.getLogger(__name__)


class NewsSource:
    @property
    def name(self) -> str:
        return "news_em"

    def _to_raw_code(self, symbol: str) -> str:
        return symbol.split(".")[1]

    @trace_akshare("stock_news_em")
    def _fetch(self, code: str) -> pd.DataFrame:
        return ak.stock_news_em(symbol=code)

    async def fetch(self, symbol: str) -> pd.DataFrame:
        code = self._to_raw_code(symbol)
        logger.info("news source: fetching %s", code)
        df = await self._fetch(code)
        if df.empty:
            logger.warning("news source: empty result for %s", code)
            return pd.DataFrame()

        rows = []
        for _, row in df.iterrows():
            publish_time = None
            if pd.notna(row.get("发布时间")):
                publish_time = pd.to_datetime(row["发布时间"]).to_pydatetime()

            rows.append({
                "symbol": symbol,
                "title": str(row.get("新闻标题", "")),
                "content": str(row.get("新闻内容", "")),
                "publish_time": publish_time,
                "source_name": str(row.get("文章来源")) if pd.notna(row.get("文章来源")) else None,
                "url": str(row.get("新闻链接", "")),
            })

        if not rows:
            return pd.DataFrame()

        result = pd.DataFrame(rows)
        result = result.sort_values("publish_time", ascending=False).reset_index(drop=True)
        logger.info("news source: returning %d records for %s", len(result), code)
        return result
```

### Task 3: Create NewsService

**Files:**
- Create: `app/news.py`

- [ ] **Step 1: Create the service file**

```python
import logging
from datetime import datetime

import asyncpg

from app.schemas import NewsRecord
from app.sources.news import NewsSource

logger = logging.getLogger(__name__)


class NewsService:
    def __init__(self):
        self.source = NewsSource()

    async def get_news(
        self,
        pool: asyncpg.Pool,
        symbol: str,
        start: datetime | None = None,
        end: datetime | None = None,
    ) -> list[NewsRecord]:
        query = "SELECT symbol, title, content, publish_time, source_name, url FROM news WHERE symbol = $1"
        params: list = [symbol]
        if start:
            query += " AND publish_time >= $" + str(len(params) + 1)
            params.append(start)
        if end:
            query += " AND publish_time <= $" + str(len(params) + 1)
            params.append(end)
        query += " ORDER BY publish_time DESC"

        async with pool.acquire() as conn:
            rows = await conn.fetch(query, *params)

        return [NewsRecord(**dict(r)) for r in rows]

    async def sync_symbol(self, pool: asyncpg.Pool, symbol: str) -> int:
        df = await self.source.fetch(symbol)
        if df.empty:
            return 0

        rows: list[tuple] = []
        for _, row in df.iterrows():
            rows.append((
                row["symbol"], row["title"], row["content"],
                row["publish_time"], row["source_name"], row["url"],
            ))

        async with pool.acquire() as conn:
            await conn.executemany(
                """INSERT INTO news (symbol, title, content, publish_time, source_name, url)
                   VALUES ($1, $2, $3, $4, $5, $6)
                   ON CONFLICT (symbol, url) DO UPDATE SET
                       title = EXCLUDED.title,
                       content = EXCLUDED.content,
                       publish_time = EXCLUDED.publish_time,
                       source_name = EXCLUDED.source_name,
                       updated_at = NOW()""",
                rows,
            )
        return len(rows)


news_service = NewsService()
```

```python
@router.get("/news/{symbol}", response_model=list[NewsRecord])
async def news(
    symbol: str = Path(pattern=r"^(SH|SZ|BJ)\.[0-9]{6}$"),
    start: datetime | None = Query(None),
    end: datetime | None = Query(None),
    refresh: bool = Query(False),
):
    logger.info("GET /news/%s refresh=%s start=%s end=%s", symbol, refresh, start, end)
    pool = await get_pool()
    if refresh:
        count = await news_service.sync_symbol(pool, symbol)
        logger.info("GET /news/%s sync completed: %d records", symbol, count)
    records = await news_service.get_news(pool, symbol, start, end)
    logger.info("GET /news/%s returned %d records", symbol, len(records))
    return records
```
